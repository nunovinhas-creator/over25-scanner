// ========================================================= // OVER 2.5 MODEL - TRACKER OPTIMIZED VERSION // ========================================================= function calculateOverScore({ prob25, over25odds, movement, confidence, btts, xgHome, xgAway, hCoach, aCoach, hForm, aForm, h2h, homeInjured, awayInjured, recommend }) { let s = 0; const sig = []; // ===================================================== // 1. MARKET MOVEMENT (BIGGEST WEIGHT) // ===================================================== if (movement === 'SHORTENING') { s += 35; sig.push({ t: '↘ SHORTENING', c: 'g' }); } else if (movement === 'DRIFTING') { s -= 25; sig.push({ t: '↗ DRIFTING', c: 'r' }); } // ===================================================== // 2. TOTAL XG // ===================================================== const xg = (xgHome != null && xgAway != null) ? (Number(xgHome) + Number(xgAway)) : null; if (xg != null && !isNaN(xg)) { if (xg >= 3.2) { s += 25; sig.push({ t: `🔥 xG Elite ${xg.toFixed(2)}`, c: 'g' }); } else if (xg >= 2.8) { s += 18; sig.push({ t: `xG Forte ${xg.toFixed(2)}`, c: 'g' }); } else if (xg >= 2.4) { s += 12; sig.push({ t: `xG OK ${xg.toFixed(2)}`, c: 'a' }); } else if (xg < 2.0) { s -= 15; sig.push({ t: 'xG baixo', c: 'r' }); } } // ===================================================== // 3. BTTS PROBABILITY // ===================================================== const bp = btts != null ? (btts > 1 ? btts : btts * 100) : null; if (bp != null && !isNaN(bp)) { if (bp >= 65) { s += 20; sig.push({ t: `BTTS Elite ${bp.toFixed(0)}%`, c: 'g' }); } else if (bp >= 55) { s += 12; sig.push({ t: `BTTS Forte ${bp.toFixed(0)}%`, c: 'g' }); } else if (bp >= 45) { s += 5; } else if (bp < 40) { s -= 15; sig.push({ t: 'BTTS fraco', c: 'r' }); } } // ===================================================== // 4. COACH PROFILES // ===================================================== const hp = hCoach?.profile || ''; const ap = aCoach?.profile || ''; if (hp === 'attacking' && ap === 'attacking') { s += 20; sig.push({ t: '⚔️ attacking + attacking', c: 'g' }); } else if ( (hp === 'attacking' && ap === 'balanced') || (ap === 'attacking' && hp === 'balanced') ) { s += 12; sig.push({ t: '⚔️ attacking + balanced', c: 'a' }); } else if ( hp === 'balanced' && ap === 'balanced' ) { s += 5; } else if ( hp === 'defensive' && ap === 'defensive' ) { s -= 20; sig.push({ t: '🛑 defensive + defensive', c: 'r' }); } // ===================================================== // 5. RECENT OFFENSIVE FORM // ===================================================== function offensiveFormScore(form) { if (!form) return 0; const wins = (form.match(/W/g) || []).length; const draws = (form.match(/D/g) || []).length; const losses = (form.match(/L/g) || []).length; let score = 0; if (wins >= 4) { score += 10; } else if (wins >= 3) { score += 6; } else if (wins <= 1) { score -= 5; } if (losses >= 3) { score -= 4; } return score; } s += offensiveFormScore(hForm); s += offensiveFormScore(aForm); // ===================================================== // 6. DEFENSIVE INJURIES // ===================================================== const injH = Array.isArray(homeInjured) ? homeInjured.filter(p => p.status === 'injured' || p.status === 'suspended' ).length : 0; const injA = Array.isArray(awayInjured) ? awayInjured.filter(p => p.status === 'injured' || p.status === 'suspended' ).length : 0; const totalInj = injH + injA; if (totalInj >= 10) { s += 15; sig.push({ t: '🤕 muitas baixas', c: 'g' }); } else if (totalInj >= 6) { s += 8; sig.push({ t: '🤕 baixas relevantes', c: 'a' }); } // ===================================================== // 7. H2H GOALS // ===================================================== if ( h2h && h2h.avg_total_goals != null ) { const avg = Number(h2h.avg_total_goals); if (!isNaN(avg)) { if (avg >= 3.2) { s += 10; sig.push({ t: `H2H ${avg.toFixed(1)}`, c: 'g' }); } else if (avg < 2.0) { s -= 8; sig.push({ t: 'H2H under', c: 'r' }); } } } // ===================================================== // 8. OVER ODDS // ===================================================== if ( over25odds != null && !isNaN(over25odds) ) { const o = Number(over25odds); if (o <= 1.65) { s += 10; sig.push({ t: `Odds fortes ${o}`, c: 'g' }); } else if (o <= 1.80) { s += 6; } else if (o >= 2.20) { s -= 10; sig.push({ t: `Odds altas ${o}`, c: 'r' }); } } // ===================================================== // 9. ML PROBABILITY (LOWER WEIGHT) // ===================================================== if ( prob25 != null && !isNaN(prob25) ) { const p = prob25 > 1 ? prob25 : prob25 * 100; if (p >= 65) { s += 5; } else if (p < 45) { s -= 5; } } // ===================================================== // 10. RECOMMENDATION BOOST // ===================================================== if (recommend) { const r = String(recommend).toLowerCase(); if ( r.includes('over') || r.includes('btts') ) { s += 5; sig.push({ t: '📈 recomendação ofensiva', c: 'g' }); } } // ===================================================== // FINAL NORMALIZATION // ===================================================== s = Math.max( 0, Math.min(100, Math.round(s)) ); // ===================================================== // FINAL LABEL // ===================================================== let level = 'LOW'; if (s >= 85) { level = 'ELITE'; } else if (s >= 70) { level = 'VERY STRONG'; } else if (s >= 55) { level = 'STRONG'; } else if (s >= 40) { level = 'MEDIUM'; } // ===================================================== // RETURN // ===================================================== return { score: s, level, signals: sig, metrics: { xg, btts: bp, injuries: totalInj, movement, profiles: `${hp}/${ap}` } }; }
